cmake_minimum_required(VERSION 3.28)
project(obs-sacn-crossfade VERSION 0.46.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "MA OBS MediaLink requires Windows x64.")
endif()
find_package(libobs 33.0.0 REQUIRED)
find_package(obs-frontend-api 33.0.0 REQUIRED)
find_package(Qt6 6.11 REQUIRED COMPONENTS Widgets Core)
set(CMAKE_AUTORCC ON)
add_library(obs-sacn-crossfade MODULE src/plugin.cpp src/cue_math.h src/video_control.h src/audio_control.h src/audio_sources.h src/ui_support.h src/monitor_ui.h src/media_actions.h src/media_panel.h src/source_settings.h src/live_audio_policy.h src/settings_policy.h src/about_secret.h src/about_ui.h data/medialink.qrc)
target_link_libraries(obs-sacn-crossfade PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Core ws2_32 iphlpapi)
target_compile_definitions(obs-sacn-crossfade PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(obs-sacn-crossfade PRIVATE /utf-8)
endif()
set_target_properties(obs-sacn-crossfade PROPERTIES PREFIX "")
install(TARGETS obs-sacn-crossfade RUNTIME DESTINATION "obs-sacn-crossfade/bin/64bit" LIBRARY DESTINATION "obs-sacn-crossfade/bin/64bit")
install(FILES data/crossfade.effect DESTINATION "obs-sacn-crossfade/data")
install(FILES data/medialink.ico DESTINATION "obs-sacn-crossfade/data")
set(MA_PACKAGE_DIR "obs-sacn-crossfade/data/ma-paket")
install(FILES ../resources/ma-paket/START_HIER.html DESTINATION "${MA_PACKAGE_DIR}")
install(FILES ../grandMA3/MA3/MA_OBS_MediaLink.xml ../grandMA3/MA3/MA_OBS_MediaLink.lua DESTINATION "${MA_PACKAGE_DIR}/grandMA3/MA3")
install(FILES ../grandMA3/README_START_HIER.md ../grandMA3/REFERENZEN.md DESTINATION "${MA_PACKAGE_DIR}/grandMA3")
install(FILES ../fixtures/MA_OBS_MediaLink_Fade.gdtf ../fixtures/MA_OBS_MediaLink_Steuerung.gdtf ../fixtures/README.md DESTINATION "${MA_PACKAGE_DIR}/fixtures")
install(FILES README.md DESTINATION "${MA_PACKAGE_DIR}/OBS")
install(FILES ../docs/WINDOWS_INSTALLER.md ../docs/WINDOWS_ZIP.md ../docs/BUILD.md DESTINATION "${MA_PACKAGE_DIR}/docs")
install(FILES ../LICENSE ../THIRD_PARTY.md DESTINATION "${MA_PACKAGE_DIR}")
